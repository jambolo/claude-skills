cmd: export NOTES_FILE="$(mktemp)"; bash src/add.sh "buy milk"; bash src/add.sh "call Bob"; wc -l < "$NOTES_FILE" | tr -d ' '

# list numbers notes
cmd: export NOTES_FILE="$(mktemp)"; bash src/add.sh "buy milk"; bash src/add.sh "call Bob"; bash src/list.sh | tr '\n' '|'
expect: 1: buy milk|2: call Bob|
# search is case-insensitive
cmd: export NOTES_FILE="$(mktemp)"; bash src/add.sh "buy milk"; bash src/add.sh "call Bob"; bash src/search.sh bob
expect: 2: call Bob
