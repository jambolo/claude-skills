include(GoogleTest)

# One test executable per test/*.cpp
file(GLOB ${PROJECT_NAME}_TEST_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/*.cpp")

foreach(source IN LISTS ${PROJECT_NAME}_TEST_SOURCES)
    get_filename_component(name ${source} NAME_WE)
    set(test_target ${PROJECT_NAME}_${name})

    add_executable(${test_target} ${source})
    target_link_libraries(${test_target}
        PRIVATE
            ${PROJECT_NAME}::${PROJECT_NAME}
            GTest::gtest_main
    )
    target_compile_definitions(${test_target}
        PRIVATE
            $<$<PLATFORM_ID:Windows>:NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE>
    )
    set_target_properties(${test_target} PROPERTIES CXX_EXTENSIONS OFF)

    gtest_discover_tests(${test_target})
endforeach()
